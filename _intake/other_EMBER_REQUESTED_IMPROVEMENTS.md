```python
from twilio.rest import Client

def notify_sms(message):
    client = Client(TWILIO_SID, TWILIO_TOKEN)
    client.messages.create(
        to=YOUR_PHONE,
        from_=TWILIO_PHONE,
        body=f"Ember: {message}"
    )
```

**Option 2: SendGrid/SMTP** (Email)
```python
# pip install sendgrid
from sendgrid import SendGridAPIClient
from sendgrid.helpers.mail import Mail

def notify_email(subject, body):
    message = Mail(
        from_email='ember@thepod.local',
        to_emails=YOUR_EMAIL,
        subject=subject,
        html_content=body
    )
    sg = SendGridAPIClient(SENDGRID_KEY)
    sg.send(message)
```

**Option 3: Pushover** (Push notifications)
```python
# pip install python-pushover
from pushover import Client

def notify_push(message):
    client = Client(USER_KEY, api_token=API_TOKEN)
    client.send_message(message, title="Ember")
```

**Recommendation**: Pushover (simple, works on phone/desktop, no SMS fees)
